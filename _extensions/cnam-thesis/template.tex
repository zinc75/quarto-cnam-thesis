% Chapter title style:
%   - very large number in red, right-aligned
%   - bold title, right-aligned
%   - light grey rule exactly as wide as the title
% \chaptitlewithline receives the title as argument (titlesec passes it
% directly after the before-code), measures its width and typesets title + rule
% in a minipage of that width, itself right-aligned via \filleft.
\newlength{\chaptitlewidth}
\newcommand{\chaptitlewithline}[1]{%
  \settowidth{\chaptitlewidth}{\Huge\bfseries#1}%
  \ifdim\chaptitlewidth>\linewidth\setlength{\chaptitlewidth}{\linewidth}\fi%
  \begin{minipage}{\chaptitlewidth}%
    \raggedleft%
    {\Huge\bfseries#1}\\[1.5pt]%
    {\color{black!25}\rule{\chaptitlewidth}{0.4pt}}%
  \end{minipage}%
}

% \setlength{\textwidth}{420pt} above only changes the LaTeX register,
% NOT \hsize (the actual TeX parameter controlling paragraph width, managed
% by geometry). Resync \textwidth to \hsize so that fancyhdr sees
% \headwidth = \textwidth and applies no lateral offset.
% Without this line, fancyhdr computes offset = (\hsize - 420pt)/2 and
% double-counts it on even twoside pages → header runs off the page.
\setlength{\textwidth}{\hsize}

% Resync lateral margins overridden by the hardcoded Cnam block above. /
% Resynchronise les marges latérales écrasées par le bloc Cnam ci-dessus.
% The block hardcodes \oddsidemargin=0pt and \evensidemargin=0pt, which works
% accidentally for the default narrow-margin layout (both ≈ 2 cm) but breaks
% twoside margin placement when comments widen the outer margin. /
% Le bloc fixe oddsidemargin=0 et evensidemargin=0, ce qui fonctionne par
% coïncidence pour les marges étroites mais casse le twoside avec marge élargie.
% Formula: margin_register = desired_margin − 1in + |hoffset|
%          (with \hoffset = −15pt hardcoded above, so +15pt here) /
% Formule : registre = marge_souhaitée − 1in + |hoffset|
$if(comments.enabled)$
\setlength{\oddsidemargin}{\dimexpr 2.5cm - 1in + 15pt\relax}   % odd  left  = 2.5 cm (inner/binding)
\setlength{\evensidemargin}{\dimexpr 4.5cm - 1in + 15pt\relax}  % even left  = 4.5 cm (outer, notes margin)
% Also restore \marginparwidth overridden by the hardcoded \setlength{\marginparwidth}{40pt} above. /
% Restaure aussi \marginparwidth écrasé par \setlength{\marginparwidth}{40pt} ci-dessus.
